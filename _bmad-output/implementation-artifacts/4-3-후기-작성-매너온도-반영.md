# Story 4.3: 후기 작성 & 매너온도 반영

Status: done

## Story

As a 거래 완료된 판매자 또는 구매자,
I want to leave a review for the other party after a trade is complete,
So that the community can build trust through manner temperature scores.

## Acceptance Criteria

1. `reviews` 테이블은 Story 1.1 초기 마이그레이션(AR-3)에서 이미 생성됨 — **Alembic 마이그레이션 불필요** (`id, post_id, writer_email, target_email, is_positive, tags[], comment, created_at`, `uq_reviews_post_writer(post_id, writer_email)` 유니크 제약)
2. 거래 확정된 양측이 `POST /api/v1/reviews`를 호출하면 후기가 저장되고 target_email의 `manner_temp`가 즉시 업데이트된다 (긍정 +1℃, 부정 -1℃, 범위 0~99℃)
3. 동일 거래에 이미 후기를 작성한 회원이 다시 시도하면 `{"detail": "이미 후기를 작성했습니다.", "code": "REVIEW_ALREADY_SUBMITTED"}` 409 에러가 반환된다
4. Web: 거래완료 후 TradeCompleteSheet "후기 작성하기" CTA가 `/reviews/[postId]?targetEmail={email}` 페이지로 이동, 긍정/부정 선택 + 태그 + 코멘트 입력 후 제출 시 성공 토스트 표시
5. Flutter: TradeCompleteSheet "후기 작성하기" 버튼이 ReviewFormScreen으로 이동, 동일한 UX 흐름
6. 후기 작성 완료 후 상대방의 MannerTempWidget에 업데이트된 온도가 반영된다

### Task 3: users/repository.py — update_manner_temp

```python
from sqlalchemy import func, update

async def update_manner_temp(db: AsyncSession, email: str, delta: float) -> None:
    stmt = (
        update(User)
        .where(User.email == email)
        .values(
            manner_temp=func.least(99.0, func.greatest(0.0, User.manner_temp + delta))
        )
    )
    await db.execute(stmt)
    await db.commit()
```

`func.least`, `func.greatest` — SQLAlchemy SQL 함수. Python-side 처리 대신 DB-side에서 원자적으로 업데이트.

### Task 4: reviews/repository.py

```python
from sqlalchemy import select
from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession

from app.features.reviews.models import Review
from app.shared.exceptions import AppException


async def get_review_by_post_writer(db: AsyncSession, post_id: int, writer_email: str) -> Review | None:
    stmt = select(Review).where(
        Review.post_id == post_id,
        Review.writer_email == writer_email,
    )
    return (await db.execute(stmt)).scalar_one_or_none()


async def create_review(
    db: AsyncSession,
    *,
    post_id: int,
    writer_email: str,
    target_email: str,
    is_positive: bool,
    tags: list[str],
    comment: str | None,
) -> Review:
    review = Review(
        post_id=post_id,
        writer_email=writer_email,
        target_email=target_email,
        is_positive=is_positive,
        tags=tags,
        comment=comment,
    )
    db.add(review)
    try:
        await db.commit()
        await db.refresh(review)
    except IntegrityError as e:
        await db.rollback()
        raise AppException(
            code="REVIEW_ALREADY_SUBMITTED",
            detail="이미 후기를 작성했습니다.",
            status_code=409,
        ) from e
    return review
```

### Task 5: reviews/service.py — submit_review

```python
from sqlalchemy.ext.asyncio import AsyncSession

from app.features.posts import repository as posts_repo
from app.features.posts.models import Post
from app.features.reviews import repository
from app.features.reviews.schemas import CreateReviewRequest, ReviewResponse
from app.features.users import repository as users_repo
from app.shared.exceptions import AppException


async def submit_review(
    db: AsyncSession,
    writer_email: str,
    request: CreateReviewRequest,
) -> ReviewResponse:
    # 1. 게시글 존재 및 거래완료 확인
    post = await posts_repo.get_post_by_id(db, request.post_id)
    if post is None:
        raise AppException(code="POST_NOT_FOUND", detail="게시글을 찾을 수 없습니다.", status_code=404)
    if post.status != '거래완료':
        raise AppException(
            code="TRADE_NOT_COMPLETED",
            detail="거래완료된 게시글에만 후기를 작성할 수 있습니다.",
            status_code=422,
        )

    # 2. writer가 거래 당사자인지 확인
    seller_email = post.seller_email
    buyer_email = post.buyer_email
    if writer_email not in (seller_email, buyer_email):
        raise AppException(
            code="REVIEW_NOT_ELIGIBLE",
            detail="해당 거래의 판매자 또는 구매자만 후기를 작성할 수 있습니다.",
            status_code=403,
        )

    # 3. 중복 후기 검사
    existing = await repository.get_review_by_post_writer(db, request.post_id, writer_email)
    if existing:
        raise AppException(
            code="REVIEW_ALREADY_SUBMITTED",
            detail="이미 후기를 작성했습니다.",
            status_code=409,
        )

    # 4. target 결정
    target_email = buyer_email if writer_email == seller_email else seller_email

    # 5. 후기 생성 + manner_temp 업데이트
    review = await repository.create_review(
        db,
        post_id=request.post_id,
        writer_email=writer_email,
        target_email=target_email,
        is_positive=request.is_positive,
        tags=request.tags,
        comment=request.comment,
    )
    delta = 1.0 if request.is_positive else -1.0
    await users_repo.update_manner_temp(db, target_email, delta)

    return ReviewResponse(
        id=review.id,
        post_id=review.post_id,
        writer_email=review.writer_email,
        target_email=review.target_email,
        is_positive=review.is_positive,
        tags=review.tags,
        comment=review.comment,
        created_at=review.created_at,
    )
```

**Import 주의**: `posts_repo.get_post_by_id`는 `Post.is_draft == False` 필터를 포함함 — 거래완료 게시글은 반드시 `is_draft=False`이므로 정상.

### Task 6: reviews/router.py

```python
from fastapi import APIRouter, Depends
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.database import get_db
from app.core.dependencies import get_current_user
from app.features.reviews import service
from app.features.reviews.schemas import CreateReviewRequest, ReviewResponse
from app.shared.exceptions import AppException

router = APIRouter()


@router.post("/", response_model=ReviewResponse, status_code=201)
async def submit_review(
    request: CreateReviewRequest,
    current_user: dict = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
) -> ReviewResponse:
    writer_email = current_user.get("sub")
    if not writer_email:
        raise AppException(code="UNAUTHORIZED", detail="유효하지 않은 토큰입니다.", status_code=401)
    return await service.submit_review(db, writer_email, request)
```
